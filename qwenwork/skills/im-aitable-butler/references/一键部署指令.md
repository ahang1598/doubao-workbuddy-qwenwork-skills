### 3.1 poll.py（跨平台版）

```python
#!/usr/bin/env python3
"""
DingTalk 消息轮询处理器 — 跨平台版
由 macOS LaunchAgent 或 Windows 计划任务定时调用，
轮询所有 target 的最新消息，按 messageId 去重，
筛选 @mention 后通过 MCP API 创建千问办公任务处理。
auth 过期时自动通过连接器 reconnect 刷新。
"""

import json
import os
import subprocess
import sys
import time

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
CONFIG_FILE = os.path.join(BASE_DIR, "config.json")
STATE_FILE = os.path.join(BASE_DIR, "poll-state.json")
PROCESSOR_PROMPT = os.path.join(BASE_DIR, "processor-prompt.md")
LOG_DIR = os.path.join(BASE_DIR, "logs")
LOG_FILE = os.path.join(LOG_DIR, "poll.log")
EMOTION_CACHE_FILE = os.path.join(BASE_DIR, "emotion-cache.json")
MCP_CONFIG_DEFAULT = os.path.join(os.path.expanduser("~"), ".qwenworkcn", "mcp-adaptor.config")

IS_WINDOWS = os.name == "nt"


def log(msg):
    ts = time.strftime("%Y-%m-%d %H:%M:%S")
    line = f"{ts} {msg}"
    print(line, flush=True)
    try:
        os.makedirs(LOG_DIR, exist_ok=True)
        with open(LOG_FILE, "a", encoding="utf-8") as f:
            f.write(line + "\n")
    except Exception:
        pass


def load_config():
    with open(CONFIG_FILE, "r", encoding="utf-8") as f:
        return json.load(f)


def get_dws_path(config):
    """从 config 读取 dws 路径，空则自动探测。"""
    path = config.get("dws_path", "")
    if path:
        return path
    home = os.path.expanduser("~")
    if IS_WINDOWS:
        candidates = [
            os.path.join(home, ".local", "bin", "dws.exe"),
            os.path.join(home, ".local", "bin", "dws.cmd"),
            os.path.join(home, ".local", "bin", "dws.bat"),
            os.path.join(home, "AppData", "Local", "Programs", "QwenWorkCN", "resources", "bin", "dws.exe"),
        ]
        which_cmd = "where"
    else:
        candidates = [os.path.join(home, ".local", "bin", "dws")]
        which_cmd = "which"
    for c in candidates:
        if os.path.exists(c):
            return c
    try:
        result = subprocess.run([which_cmd, "dws"],
                                capture_output=True, text=True, timeout=5, shell=IS_WINDOWS)
        if result.returncode == 0 and result.stdout.strip():
            return result.stdout.strip().splitlines()[0].strip()
    except Exception:
        pass
    return "dws"


def get_mcp_config_path(config):
    path = config.get("mcp_config_path", "")
    return path if path else MCP_CONFIG_DEFAULT


def load_state():
    try:
        with open(STATE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except (FileNotFoundError, json.JSONDecodeError):
        return {}


def save_state(state):
    with open(STATE_FILE, "w", encoding="utf-8") as f:
        json.dump(state, f, ensure_ascii=False, indent=2)


def run_dws(dws_path, args):
    """Run dws command. Windows 下 .cmd/.bat 或裸命令名需要 shell=True。"""
    cmd = [dws_path] + args + ["--format", "json", "-y"]
    use_shell = IS_WINDOWS and (dws_path.lower().endswith((".cmd", ".bat")) or dws_path == "dws")
    try:
        if use_shell:
            result = subprocess.run(" ".join(f'"{c}"' if " " in c else c for c in cmd),
                                    capture_output=True, text=True, timeout=30, shell=True)
        else:
            result = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
        if result.returncode != 0:
            return None
        return json.loads(result.stdout)
    except Exception as e:
        log(f"  dws error: {e}")
        return None


def mcp_call(mcp_config_path, method, params=None, timeout=10):
    """Call QwenWork local MCP API."""
    try:
        with open(mcp_config_path) as f:
            cfg = json.load(f)
        url = cfg["url"].rstrip("/") + "/mcp"
        headers = {"Content-Type": "application/json", "Accept": "application/json, text/event-stream"}
        headers.update(cfg.get("headers", {}))
        body = {"jsonrpc": "2.0", "id": 1, "method": method}
        if params:
            body["params"] = params

        import urllib.request as urlreq
        req = urlreq.Request(
            url, data=json.dumps(body).encode("utf-8"),
            headers=headers, method="POST",
        )
        with urlreq.urlopen(req, timeout=timeout) as resp:
            raw = resp.read().decode("utf-8")

        if "data:" in raw:
            for line in raw.split("\n"):
                if line.startswith("data:") and "{" in line:
                    resp_data = json.loads(line[5:].strip())
                    break
            else:
                resp_data = json.loads(raw)
        else:
            resp_data = json.loads(raw)

        result = resp_data.get("result", {})
        if isinstance(result, dict) and "content" in result:
            content = result["content"]
            if isinstance(content, list) and len(content) > 0:
                text = content[0].get("text", "")
                if text:
                    try:
                        return json.loads(text)
                    except (json.JSONDecodeError, ValueError):
                        return {"text": text}
        return result if isinstance(result, dict) else {"text": str(result)}
    except Exception as e:
        log(f"  mcp error: {e}")
        return None


def get_tz_offset_str():
    """自动取本机时区偏移（如 +08:00）。"""
    off = time.timezone if time.daylight == 0 else time.altzone
    off = -off
    sign = "+" if off >= 0 else "-"
    hours = abs(off) // 3600
    minutes = (abs(off) % 3600) // 60
    return f"{sign}{hours:02d}:{minutes:02d}"


def check_and_refresh_auth(dws_path, mcp_config_path):
    """Check DWS auth status only — do NOT auto-refresh.
    dws CLI renews the access_token natively via refresh_token when it expires,
    provided the keychain holds a proper browser-OAuth token. Running the
    QwenWork connector disconnect→connect REPLACES that token with an
    MCP authCode-based one whose refresh always fails — this is the root
    cause of repeated auth breakage. So we never touch the connector,
    and dws auth login always needs a browser, so we never call it here."""
    result = run_dws(dws_path, ["auth", "status"])
    if result and result.get("authenticated") and result.get("token_valid"):
        return True
    reason = result.get("reason", "unknown") if result else "no response"
    log(f"  auth not available ({reason}) — user needs to run 'dws auth login' in terminal")
    log("  NOTE: never disconnect/reconnect the QwenWork DWS connector; it breaks token refresh")
    return False


def load_emotion_cache():
    try:
        with open(EMOTION_CACHE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except (FileNotFoundError, json.JSONDecodeError):
        return {}


def save_emotion_cache(cache):
    with open(EMOTION_CACHE_FILE, "w", encoding="utf-8") as f:
        json.dump(cache, f, ensure_ascii=False, indent=2)


def ensure_emotion(dws_path, config, stage):
    cache = load_emotion_cache()
    emo = cache.get(stage)
    if emo and emo.get("emotion_id"):
        return emo
    emotions_cfg = config.get("emotions", {})
    stage_cfg = emotions_cfg.get(stage, {})
    emotion_name = stage_cfg.get("name", stage)
    text = stage_cfg.get("text", stage)
    background_id = stage_cfg.get("background_id", "im_bg_3")
    result = run_dws(dws_path, [
        "chat", "message", "create-text-emotion",
        "--emotion-name", emotion_name, "--text", text,
        "--background-id", background_id,
    ])
    if result and result.get("emotionId"):
        emo = {"emotion_id": result["emotionId"], "emotion_name": emotion_name,
               "text": text, "background_id": background_id}
        cache[stage] = emo
        save_emotion_cache(cache)
        return emo
    return None


def add_text_emotion(dws_path, config, conv_id, msg_id, stage):
    emo = ensure_emotion(dws_path, config, stage)
    if not emo or not emo.get("emotion_id"):
        return
    run_dws(dws_path, [
        "chat", "message", "add-text-emotion",
        "--conversation-id", conv_id, "--msg-id", msg_id,
        "--emotion-id", emo["emotion_id"],
        "--emotion-name", emo.get("emotion_name", stage),
        "--text", emo.get("text", stage),
        "--background-id", emo.get("background_id", "im_bg_3"),
    ])


def poll_target(target, config, state, seen_ids, dws_path, mcp_config_path):
    name = target.get("name", "?")
    conv_id = target.get("conversation_id", "")
    if not conv_id or not target.get("enabled", True):
        return

    target_key = f"{name}_last_time"
    last_time = state.get(target_key, "")

    if target.get("type") == "direct":
        user_id = target.get("user_id", "")
        if not user_id:
            return
        from datetime import datetime, timedelta
        start_time = last_time if last_time else (datetime.now() - timedelta(hours=1)).strftime("%Y-%m-%d %H:%M:%S")
        data = run_dws(dws_path, ["chat", "+messages-list-direct", "--user", user_id, "--time", start_time, "--limit", "10"])
    else:
        data = run_dws(dws_path, ["chat", "+chat-messages", "--id", conv_id, "--limit", "10"])
    if not data or not data.get("messages"):
        return

    keywords = config.get("keywords", [])
    strict_default = config.get("strict_at_check", True)
    bots = config.get("bots", {})
    bot_priority = config.get("bot_priority", list(bots.keys()))
    default_bot = config.get("default_bot", bot_priority[0] if bot_priority else "")
    self_bot_names = config.get("self_bot_names", [])
    max_age_minutes = config.get("message_max_age_minutes", 10)

    new_count = 0
    for msg in data["messages"]:
        msg_id = msg.get("messageId", "")
        if not msg_id or msg_id in seen_ids:
            continue
        seen_ids.add(msg_id)

        sender = msg.get("sender") or msg.get("senderId", "")
        content = msg.get("text", "")
        create_time = msg.get("createTime", "")

        if self_bot_names and sender in self_bot_names:
            continue
        if last_time and create_time <= last_time:
            continue

        try:
            from datetime import datetime, timedelta
            msg_time = datetime.strptime(create_time, "%Y-%m-%d %H:%M:%S")
            if datetime.now() - msg_time > timedelta(minutes=max_age_minutes):
                continue
        except (ValueError, TypeError):
            pass

        strict = target.get("strict_at_check", strict_default)
        if strict:
            has_at = any(f"@{kw}" in content for kw in keywords)
            if not has_at:
                continue
            route_to = default_bot
            for bot_name in bot_priority:
                if f"@{bot_name}" in content:
                    route_to = bot_name
                    break
        else:
            route_to = target.get("default_route") or default_bot

        bot_cfg = bots.get(route_to, {})
        reply_mode = bot_cfg.get("reply_mode", "markdown")
        robot_code = bot_cfg.get("robot_code")

        target_type = target.get("type", "group")
        target_id = target.get("user_id", conv_id) if target_type == "direct" else conv_id
        msg_conv_id = msg.get("conversationId", conv_id)

        log(f"  [{name}] from '{sender}' → '{route_to}' ({reply_mode}): {content[:80]}")
        add_text_emotion(dws_path, config, msg_conv_id, msg_id, "working")

        archive_cfg = config.get("archive", {})
        archive_line = (
            f"存档AI表格文档ID: {archive_cfg.get('aitable_doc_id')}\n"
            f"存档子表名: {archive_cfg.get('sheet_name', '问题收集表')}\n"
            if archive_cfg.get("enabled") else "存档: 不启用\n"
        )
        prompt = (
            f"请阅读文件 {PROCESSOR_PROMPT} 获取处理指令，"
            f"然后处理以下钉钉消息：\n\n"
            f"## 消息数据\n"
            f"发送人: {sender}\n"
            f"群/会话ID: {msg_conv_id}\n"
            f"消息ID: {msg_id}\n"
            f"目标类型: {target_type}\n"
            f"目标ID: {target_id}\n"
            f"群名: {name}\n"
            f"消息时间: {create_time}\n"
            f"路由目标: {route_to}\n"
            f"回复模式: {reply_mode}\n"
            f"机器人代码: {robot_code or 'N/A'}\n"
            f"{archive_line}\n"
            f"## 消息内容\n{content}"
        )

        try:
            job = {
                "name": "msg-%d" % int(time.time()),
                "schedule": {"kind": "at", "at": time.strftime("%Y-%m-%dT%H:%M:%S", time.localtime(time.time() + 3600)) + get_tz_offset_str()},
                "payload": {"kind": "agentTurn", "message": prompt, "contextDirs": [BASE_DIR]},
            }
            r = mcp_call(mcp_config_path, "tools/call", {"name": "qwenwork_cron_manage_job", "arguments": {"action": "add", "job": job}}, timeout=15)
            job_id = r.get("taskId") if r else None
            if job_id:
                mcp_call(mcp_config_path, "tools/call", {"name": "qwenwork_cron_manage_job", "arguments": {"action": "run", "jobId": job_id}}, timeout=15)
                mcp_call(mcp_config_path, "tools/call", {"name": "qwenwork_cron_manage_job", "arguments": {"action": "remove", "jobId": job_id}}, timeout=10)
                log(f"  Triggered MCP task → '{route_to}' ({reply_mode})")
            else:
                log(f"  MCP task creation failed: {r}")
        except Exception as e:
            log(f"  MCP task trigger failed: {e}")

        new_count += 1

    if data["messages"]:
        latest_time = max(m.get("createTime", "") for m in data["messages"])
        if latest_time > last_time:
            state[target_key] = latest_time
    if new_count:
        log(f"  [{name}] {new_count} new message(s) processed")


def main():
    log("=== Poll started ===")
    config = load_config()
    dws_path = get_dws_path(config)
    mcp_config_path = get_mcp_config_path(config)

    if not check_and_refresh_auth(dws_path, mcp_config_path):
        log("  auth not available, skipping this poll cycle")
        log("=== Poll done (auth failed) ===")
        return

    state = load_state()
    seen_ids = set(state.get("seen_ids", []))

    for target in config.get("targets", []):
        if not target.get("enabled", True):
            continue
        try:
            poll_target(target, config, state, seen_ids, dws_path, mcp_config_path)
        except Exception as e:
            log(f"  [{target.get('name', '?')}] error: {e}")

    retention = config.get("seen_ids_retention", 500)
    state["seen_ids"] = list(seen_ids)[-retention:]
    save_state(state)
    log("=== Poll done ===")


if __name__ == "__main__":
    main()
```

### 3.2 processor-prompt.md

创建 `processor-prompt.md`，内容如下（`{{BASE_DIR}}` 替换为安装目录绝对路径，macOS 用正斜杠、Windows 用反斜杠）：

```
# 钉钉 @mention 消息处理指令

你收到了一条包含 @mention 的钉钉消息。请严格按照以下步骤处理。

## 0. 闲聊过滤（仅 card 模式）

仅对 card 模式（机器人代码为 N/A）的消息执行此步骤。markdown 模式跳过此步。

闲聊特征（满足任一）：纯打招呼/寒暄、感谢/客套、简单确认（好的/收到/ok）、纯表情、与工作无关的日常对话。
问题特征：技术标识（taskId/traceId/报错/失败）、求助语句（帮忙看看/为什么/怎么）、具体技术问题或业务需求。

闲聊消息：不创建卡片、不发送任何回复，只走步骤3更新表情为"处理完成"后结束。
问题消息：正常进入步骤1。

## 1. 理解消息内容

- 去掉 @mention 前缀，只处理实际问题部分
- 忽略图片链接和文档链接
- 如需联网搜索最新信息，使用 WebSearch
- 生成简洁专业的回复（Markdown 格式）

## 2. 根据回复模式发送

注意：所有 dws 命令通过 Bash 工具运行时，若直接 `dws` 不识别（Windows 常见），改用完整路径（从 config.json 的 dws_path 字段读取）。

### markdown 模式（机器人代码有值）：
- 群聊: dws chat +messages-send-by-bot --robot-code "<robot_code>" --group "<conversation_id>" --title "回复" --text "<回复内容>" --yes --format json
- 单聊: dws chat +messages-send-by-bot --robot-code "<robot_code>" --user "<target_id>" --title "回复" --text "<回复内容>" --yes --format json

### card 模式（机器人代码为 N/A）：
1. 创建卡片：dws chat message send-card --group "<conversation_id>" -f json -y（单聊用 --receiver），从返回 JSON 获取 bizId
2. 更新"处理中"：dws chat message update-card --biz-id "<bizId>" --content "## <route_to> · 处理中..." --flow-status 2 -f json -y
3. 等1秒，更新最终内容 flow-status=2，再等0.8秒更新 flow-status=3（最后一步必须是3，否则卡片一直显示"生成中"）
4. send-card 失败则降级为 messages-send-by-bot 发纯文本

## 3. 更新原消息文字表情
```

### Windows — 任务计划程序（schtasks）

Windows 计划任务最小重复间隔为 1 分钟，用方案 A 即可（30 秒延迟可接受）；需要 30 秒间隔用方案 B：

方案A（每 1 分钟）：

```
schtasks /Create /TN "QwenWorkDingTalkPoller" /TR "\"<PYTHON_PATH>\" \"<BASE_DIR>\poll.py\"" /SC MINUTE /MO 1 /F
```

方案B（30 秒）——先创建 wrapper 批处理 `<BASE_DIR>\poll_twice.bat`：

```bat
@echo off
"<PYTHON_PATH>" "<BASE_DIR>\poll.py"
timeout /t 30 /nobreak >nul
"<PYTHON_PATH>" "<BASE_DIR>\poll.py"
```

再注册：

```
schtasks /Create /TN "QwenWorkDingTalkPoller" /TR "\"<BASE_DIR>\poll_twice.bat\"" /SC MINUTE /MO 1 /F
```

要点：`<PYTHON_PATH>` 用第一步检测到的 python.exe 完整路径；创建后 `schtasks /Run /TN "QwenWorkDingTalkPoller"` 手动触发一次验证；创建失败提示权限问题时不要用 /RL HIGHEST，普通权限即可。
