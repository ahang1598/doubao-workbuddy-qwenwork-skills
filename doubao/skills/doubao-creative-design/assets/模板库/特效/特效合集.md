```javascript
/* 特效-物体投影
 * 一套输入契约适配鞋、瓶、耳机、戒指等不同轮廓。
 * 页面位置由构图决定；本文件只在主体局部坐标中计算投射影和接触影。
 * 输出是独立 RGBA Canvas，放在承载面之上、主体之下。
 */
(function (global) {
  "use strict";

  const DEFAULTS = Object.freeze({
    projection: [1, 0, -0.38, 0.26, 0.08, 0.62],
    castBlur: 0.018,
    castOpacity: 0.16,
    contactBlur: 0.006,
    contactOpacity: 0.24,
    shadowColor: "#000000",
    blendMode: "source-over"
  });

  function makeCanvas(width, height) {
    const node = document.createElement("canvas");
    node.width = Math.max(1, Math.round(width));
    node.height = Math.max(1, Math.round(height));
    return node;
  }

  function sourceSize(source) {
    return {
      width: source.naturalWidth || source.videoWidth || source.width,
      height: source.naturalHeight || source.videoHeight || source.height
    };
  }

  function colorizeAlpha(target, color, opacity) {
    const ctx = target.getContext("2d");
    ctx.globalCompositeOperation = "source-in";
    ctx.globalAlpha = opacity;
    ctx.fillStyle = color;
    ctx.fillRect(0, 0, target.width, target.height);
    ctx.globalAlpha = 1;
    ctx.globalCompositeOperation = "source-over";
  }

  function drawContactRegions(target, contacts, scale, blur, color, opacity) {
    if (!contacts || !contacts.length || opacity <= 0) return;
    const raw = makeCanvas(target.width, target.height);
    const rctx = raw.getContext("2d");
    rctx.fillStyle = "#000";
    rctx.strokeStyle = "#000";
    rctx.lineCap = "round";
    for (const item of contacts) {
      rctx.globalAlpha = Math.max(0, Math.min(1, Number(item.weight ?? 1)));
      if (item.type === "line") {
        rctx.lineWidth = Math.max(1, Number(item.width || 0.02) * scale);
        rctx.beginPath();
        rctx.moveTo(item.x1 * target.width, item.y1 * target.height);
        rctx.lineTo(item.x2 * target.width, item.y2 * target.height);
        rctx.stroke();
      } else {
        const rx = Number(item.rx || item.radius || 0.02) * target.width;
        const ry = Number(item.ry || item.radius || 0.012) * target.height;
        rctx.beginPath();
        rctx.ellipse(item.x * target.width, item.y * target.height, rx, ry, 0, 0, Math.PI * 2);
        rctx.fill();
      }
    }
    const ctx = target.getContext("2d");
    ctx.save();
    ctx.filter = `blur(${Math.max(0, blur * scale)}px)`;
    ctx.globalAlpha = opacity;
    ctx.drawImage(raw, 0, 0);
    ctx.restore();
    colorizeAlpha(target, color, 1);
  }

  /**
   * source: 含 alpha 的主体。
   * projection: Canvas 2D 矩阵 [a,b,c,d,e,f]，e/f 为画布宽高比例。
   * contacts: 点／线接触区，值相对输出画布归一化。
   * receiverMask: 可选承载面 alpha，用于限制投影范围。
   */
  function renderObjectShadow(input) {
    if (!input || !input.source) throw new TypeError("source is required");
    const size = sourceSize(input.source);
    if (!size.width || !size.height) throw new Error("source has no drawable size");
    const width = input.width || size.width;
    const height = input.height || size.height;
    const scale = Math.max(width, height);
    const out = makeCanvas(width, height);
    const cast = makeCanvas(width, height);
    const cctx = cast.getContext("2d");
    const [a, b, c, d, e, f] = (input.projection || DEFAULTS.projection).map(Number);
    cctx.save();
    cctx.filter = `blur(${Math.max(0, Number(input.castBlur ?? DEFAULTS.castBlur) * scale)}px)`;
    cctx.setTransform(a, b, c, d, e * width, f * height);
    cctx.drawImage(input.source, 0, 0, width, height);
    cctx.restore();
    colorizeAlpha(cast, input.shadowColor || DEFAULTS.shadowColor, Number(input.castOpacity ?? DEFAULTS.castOpacity));

    const ctx = out.getContext("2d");
    ctx.globalCompositeOperation = input.blendMode || DEFAULTS.blendMode;
    ctx.drawImage(cast, 0, 0);
    const contact = makeCanvas(width, height);
    drawContactRegions(
      contact,
      input.contacts || [],
      scale,
      Number(input.contactBlur ?? DEFAULTS.contactBlur),
      input.shadowColor || DEFAULTS.shadowColor,
      Number(input.contactOpacity ?? DEFAULTS.contactOpacity)
    );
    ctx.drawImage(contact, 0, 0);
    if (input.receiverMask) {
      ctx.globalCompositeOperation = "destination-in";
      ctx.drawImage(input.receiverMask, 0, 0, width, height);
    }
    ctx.globalCompositeOperation = "source-over";
    return out;
  }

  global.ObjectShadowEffect = Object.freeze({ defaults: DEFAULTS, render: renderObjectShadow });
})(globalThis);
```
