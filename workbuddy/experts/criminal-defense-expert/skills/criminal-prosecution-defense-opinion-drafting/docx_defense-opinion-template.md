## python-docx 关键代码片段

```python
from docx import Document
from docx.shared import Pt, Cm, Emu, DXA
from docx.enum.text import WD_ALIGN_PARAGRAPH

def create_defense_opinion():
    doc = Document()
    
    # §1 页面布局
    section = doc.sections[0]
    section.page_width = Cm(21.0)
    section.page_height = Cm(29.7)
    section.top_margin = Cm(3.7)    # 2100 DXA
    section.bottom_margin = Cm(3.5) # 1984 DXA
    section.left_margin = Cm(2.8)   # 1588 DXA
    section.right_margin = Cm(2.6)  # 1474 DXA
    
    # §1.3 行距
    from docx.shared import Twips
    from docx.oxml.ns import qn
    for paragraph in doc.paragraphs:
        pPr = paragraph._p.get_or_add_pPr()
        spacing = pPr.find(qn('w:spacing'))
        if spacing is None:
            spacing = OxmlElement('w:spacing')
            pPr.append(spacing)
        spacing.set(qn('w:line'), '560')  # 28磅固定行距
        spacing.set(qn('w:lineRule'), 'exact')
    
    # §2 标题
    title = doc.add_paragraph()
    title.alignment = WD_ALIGN_PARAGRAPH.CENTER
    run = title.add_run('关于[嫌疑人姓名]涉嫌[罪名]的辩护意见书')
    run.font.name = '方正小标宋简体'
    run.font.size = Pt(22)
    run.bold = True
    
    # §9.2 致送机关
    salutation = doc.add_paragraph()
    salutation.alignment = WD_ALIGN_PARAGRAPH.LEFT
    run = salutation.add_run('[XX]人民检察院：')
    run.font.name = '仿宋_GB2312'
    run.font.size = Pt(16)
    # 注意：顶格，不加缩进
    
    # §9.3 前言
    intro = doc.add_paragraph()
    intro.paragraph_format.first_line_indent = Cm(0.64)  # 2字符≈640DXA
    run = intro.add_run(
        '辩护人[律师姓名]，[律所名称]律师，执业证号[XXX]，'
        '系犯罪嫌疑人[姓名]涉嫌[罪名]一案的辩护人。'
        '辩护人已查阅全部侦查卷宗，经认真研究，'
        '现依据《中华人民共和国刑事诉讼法》第173条之规定，'
        '提出如下辩护意见：'
    )
    run.font.name = '仿宋_GB2312'
    run.font.size = Pt(16)
    
    # 后续段落按 docx-format-spec.md 逐步渲染...
    
    return doc
```
